# To Do: Create a visualization that will support your conclusions for part two


# Create output folder if it does not already exist
os.makedirs("output", exist_ok=True)

# Clean the ridership column so it can be used in calculations
filtered_data["Unlinked Passenger Trips FY"] = (
    filtered_data["Unlinked Passenger Trips FY"]
    .astype(str)
    .str.replace(",", "", regex=False)
    .str.replace("$", "", regex=False)
    .str.strip()
)
filtered_data["Unlinked Passenger Trips FY"] = pd.to_numeric(
    filtered_data["Unlinked Passenger Trips FY"], errors="coerce"
)

# Clean the year column
filtered_data["Last Closed FY End Year"] = pd.to_numeric(
    filtered_data["Last Closed FY End Year"], errors="coerce"
)

# Group data by year and agency, then sum ridership
line_data = filtered_data.groupby(
    ["Last Closed FY End Year", "Agency"]
)["Unlinked Passenger Trips FY"].sum().reset_index()

# Create the line chart
plt.figure(figsize=(10, 6))
for agency in neighborhoods:
    agency_data = line_data[line_data["Agency"] == agency]
    plt.plot(
        agency_data["Last Closed FY End Year"],
        agency_data["Unlinked Passenger Trips FY"],
        marker="o",
        label=agency
    )

# Add title and labels
plt.title("Ridership Trends Over Years")
plt.xlabel("Year")
plt.ylabel("Unlinked Passenger Trips")
plt.legend()
plt.xticks(rotation=45)
plt.tight_layout()

# Save and display the chart
plt.savefig("output/visualization1.png")
plt.close()
display(Image(filename="output/visualization1.png"))
