    max_clusters = 4

    # Ward's method hierarchical clustering
    linkage_matrix = linkage(normalized_data, method="ward")

print("Creating feature importance heatmap to show mean feature values by cluster...")
# To Do: Create a feature importance heatmap, and save it into a png file under the 'output' folder.

try:
    # Create a working dataframe with cluster labels
    clustering_df = data_for_clustering.copy()
    clustering_df["Cluster"] = cluster_labels.astype(int)

    # Mean feature values by cluster
    cluster_means = clustering_df.groupby("Cluster").mean(numeric_only=True)

    # Keep the most variable features for readability
    feature_variance = data_for_clustering.var().sort_values(ascending=False)
    top_features = feature_variance.head(15).index.tolist()

    heatmap_data = cluster_means[top_features]

    plt.figure(figsize=(12, 6))
    sns.heatmap(heatmap_data, cmap="viridis", annot=False)
    plt.title("Feature Importance Heatmap (Cluster Mean Values)")
    plt.xlabel("Features (Top 15 by Variance)")
    plt.ylabel("Cluster")

    heatmap_path = "output/heatmap.png"
    plt.tight_layout()
    plt.savefig(heatmap_path, dpi=150)
    plt.close()

    print(f"Saved heatmap to: {heatmap_path}")
